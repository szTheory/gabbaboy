cmake_minimum_required(VERSION 3.25)
project(GabbaBoyTimeoutPolicyProbe NONE)
enable_testing()

if(NOT DEFINED GBB_TIMEOUT_HELPER OR NOT EXISTS "${GBB_TIMEOUT_HELPER}")
  message(FATAL_ERROR "GBB_TIMEOUT_HELPER must name the shared timeout policy")
endif()
if(NOT DEFINED GBB_POLICY_SCENARIO OR
   NOT GBB_POLICY_SCENARIO MATCHES "^(complete|normal|early|double|parent-only)$")
  message(FATAL_ERROR "GBB_POLICY_SCENARIO must be complete, normal, early, double, or parent-only")
endif()
if(NOT DEFINED GBB_POLICY_SANITIZERS)
  message(FATAL_ERROR "GBB_POLICY_SANITIZERS must be set")
endif()

set(GABBABOY_ENABLE_SANITIZERS "${GBB_POLICY_SANITIZERS}")
include("${GBB_TIMEOUT_HELPER}")
file(WRITE "${CMAKE_BINARY_DIR}/timeout-properties.txt" "")

function(gbb_policy_add_test name timeout)
  add_test(NAME "${name}" COMMAND "${CMAKE_COMMAND}" -E true)
  set_tests_properties("${name}" PROPERTIES TIMEOUT "${timeout}")
endfunction()

gbb_policy_add_test(initial_120 120)
gbb_policy_add_test(age_240 240)
gbb_policy_add_test(partition_60 60)

if(GBB_POLICY_SCENARIO STREQUAL "early")
  gbb_scale_sanitizer_test_timeouts()
  gbb_policy_add_test(late_120 120)
else()
  gbb_policy_add_test(late_120 120)
endif()

add_subdirectory(child)
if(NOT GBB_POLICY_SCENARIO STREQUAL "early")
  gbb_scale_sanitizer_test_timeouts()
  if(GBB_POLICY_SCENARIO STREQUAL "double")
    gbb_scale_sanitizer_test_timeouts()
  endif()
endif()

foreach(gbb_test IN ITEMS initial_120 age_240 partition_60 late_120)
  get_test_property(${gbb_test} TIMEOUT gbb_timeout)
  file(APPEND "${CMAKE_BINARY_DIR}/timeout-properties.txt" "${gbb_test}=${gbb_timeout}\n")
endforeach()
